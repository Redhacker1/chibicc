cmake_policy(SET CMP0076 NEW)
cmake_minimum_required( VERSION 3.10 )
project( chibicc C )
set( CMAKE_C_STANDARD 11 )

add_executable(chibicc-cli
	main.c
	hashmap.c
	parse.c
	preprocess.c
	strings.c
	tokenize.c
	type.c
	unicode.c
	ir/ir.c
	ir/opt.c
	ir/regalloc.c
	codegen/codegen.c
	codegen/common/common.c
	codegen/x86_64/x86_64.c
	codegen/m68k/m68k.c
	codegen/z80/z80.c
	abi/abi.c
	abi/objfmt.c
	abi/sysV64/sysv64.c
	abi/win64/win64.c
	abi/win32/win32.c
	abi/sys6/sys6.c
	abi/z80/z80.c
)

target_include_directories(chibicc-cli PRIVATE compiler_include ${CMAKE_CURRENT_LIST_DIR})

if(WIN32)

target_sources(chibicc-cli PUBLIC win_src/glob.c win_src/wait.c win_src/win_shims.c win_src/unistd.c)

target_include_directories(chibicc-cli PRIVATE win_includes)

if(MSVC)
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} /SUBSYSTEM:CONSOLE /DYNAMICBASE /NXCOMPAT")
    set(CMAKE_EXE_LINKER_FLAGS_RELEASE "${CMAKE_EXE_LINKER_FLAGS_RELEASE} /OPT:REF /OPT:ICF /INCREMENTAL:NO")
    set(CMAKE_EXE_LINKER_FLAGS_DEBUG "${CMAKE_EXE_LINKER_FLAGS_DEBUG} /INCREMENTAL")
endif()
endif()

add_custom_target(copy_headers ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${CMAKE_BINARY_DIR}/sdk/include
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_CURRENT_SOURCE_DIR}/include
        ${CMAKE_BINARY_DIR}/include
)
add_dependencies(chibicc-cli copy_headers)
